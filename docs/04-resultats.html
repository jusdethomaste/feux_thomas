La méthode de Serra (Serra et al. 2010)fait une synthèse assez complète des concepts vus au chapitre 2. D’abord, on regarde les points chauds d’une région au temps \(t=0\) (voir Figure 4.1),

Ensuite, on génère un processus de Poisson spatial selon la fonction \(\theta (x,y)=k\times f_w(x,y)\), où \(k\) est le coefficient saisonnier (par exemple, lors d’une sécheresse ou d’une canicule, \(k\) est plus grand, là où lors d’un grand froid, \(k\) est petit) et \(f_w(x,y)\) est la fonction d’intensité du total de combustible(voir Figure 4.3).Il est naturel de faire en sorte que \(k\) dépende du temps, étant donné qu’il s’agit d’un coefficient saisonnier, de telle sorte que \(\theta (x,y)=k\times f_w(x,y)\) est remplacée par \(\theta (x,y,t)=k(t)\times f_w(x,y)\).

4.1.1 Exemple 1 (\(\theta\) et \(\delta\) moyens)

Dans les images ci-dessous, un exemple a été réalisé avec la fonction d’intensité \(\theta_1(x,y)=100+100x+100y+100y^3\) (à noter que k est constante dans ce cas-ci) et \(\delta_1(x,y)=\frac{\lfloor 4+3x+2y\rfloor}{100}\). Une modification a été ajoutée à l’idée de base; les feux ne peuvent pas se propager à un endroit qui a été brûlé depuis un certain temps (2 jours dans ce cas-ci).

Code 
library(spatstat)
library(ggplot2)
library(ggforce)
library(grid)

set.seed(40)

#--------------------------
# INITIALISATION
#--------------------------

w <- owin(c(0,1), c(0,1))
aire_fenetre <- spatstat.geom::area(w)

hotspot <- function(x,y){
   6*x + 4*y
}

# --- THETA : Intensité des combustibles ---
theta <- function(x,y){
  ( 100 + 100*x + 100*y^3 + 100*y)
}

# --- DELTA : Dilatation géométrique déterministe ---
delta <- function(x, y) {
   floor(4 + 3 * x + 2 * y) / 100
}

# Feux initiaux (Jour 1)
pts_actuels <- rpoispp(hotspot, win=w)

# Structures pour l'historique et les graphiques
historique_cercles_total <- data.frame()
union_totale <- emptywindow(w)

suivi_mesures <- data.frame(
   jour = integer(),
   fraction_aire = numeric(),
   frontiere_interne = numeric(),
   frontiere_externe = numeric() # <-- NOUVEAU
)

jour <- 1

#--------------------------
# BOUCLE PRINCIPALE
#--------------------------

while(TRUE){
   
   if(pts_actuels$n==0){
      cat("Plus aucun feu actif.\n")
      break
   }
   
   # Calcul déterministe des rayons selon la position
   r <- delta(pts_actuels$x, pts_actuels$y)
   
   cercles_du_jour <- data.frame(
      x0 = pts_actuels$x,
      y0 = pts_actuels$y,
      r  = r,
      jour = jour
   )
   
   historique_cercles_total <- rbind(historique_cercles_total, cercles_du_jour)
   
   # Union des disques du jour
   liste_disques <- vector("list", pts_actuels$n)
   for(i in seq_len(pts_actuels$n)){
      liste_disques[[i]] <- disc(r[i], centre=c(pts_actuels$x[i], pts_actuels$y[i]))
   }
   
   union_du_jour <- intersect.owin(union.owin(as.solist(liste_disques)), w)
   
   #------------------------------------------------------
   # CORRECTION : Le feu ne se propage que sur le combustible restant
   #------------------------------------------------------
   # Zone encore intacte au début de la journée
   zone_combustible_restant <- setminus.owin(w, union_totale)
   
   # Processus de Poisson généré UNIQUEMENT dans la zone non brûlée
   pts_comb_tous <- rpoispp(theta, win = zone_combustible_restant)
   
   # Parmi les combustibles restants, lesquels sont touchés aujourd'hui ?
   retenus <- inside.owin(pts_comb_tous$x, pts_comb_tous$y, union_du_jour)
   
   pts_comb_du_jour <- ppp(pts_comb_tous$x[retenus], pts_comb_tous$y[retenus], window=w)
   pts_comb_hors_zone <- ppp(pts_comb_tous$x[!retenus], pts_comb_tous$y[!retenus], window=w)
   
   # Mise à jour de la zone brûlée globale
   union_totale <- union.owin(union_totale, union_du_jour)
   
   # --- CALCUL DES MESURES GÉOMÉTRIQUES DU CUMUL ---
   fraction_aire_cumulee <- spatstat.geom::area(union_totale) / aire_fenetre
   
   # 1. Front de flamme interne (sans les bordures du carré w)
   tous_les_contours <- edges(union_totale)
   w_interne <- erosion(w, r = 1e-6)
   contours_internes <- clip.psp(tous_les_contours, window = w_interne)
   frontiere_interne_totale <- sum(lengths_psp(contours_internes)) / aire_fenetre
   
   # 2. Frontière externe globale (incluant les bordures du carré w) - NOUVEAU
   frontiere_externe_totale <- spatstat.geom::perimeter(union_totale) / aire_fenetre
   
   # Sauvegarde des statistiques
   suivi_mesures <- rbind(suivi_mesures, data.frame(
      jour = jour,
      fraction_aire = fraction_aire_cumulee,
      frontiere_interne = frontiere_interne_totale,
      frontiere_externe = frontiere_externe_totale # <-- NOUVEAU
   ))
   
   #------------------------------------------------------
   # Préparation des données pour le plot du jour
   #------------------------------------------------------
   df_combustibles <- as.data.frame(pts_comb_du_jour)
   df_hors_zone <- as.data.frame(pts_comb_hors_zone)
   
   cercles_noirs <- historique_cercles_total[historique_cercles_total$jour < jour, ]
   
   p <- ggplot()
   
   if(nrow(cercles_noirs) > 0){
      p <- p + geom_circle(data=cercles_noirs, aes(x0=x0,y0=y0,r=r), fill="black", alpha=0.15, colour=NA)
   }
   
   p <- p +
      geom_circle(data=cercles_du_jour, aes(x0=x0,y0=y0,r=r), fill="magenta", alpha=0.3, colour="red", linewidth=0.3) +
      geom_point(data=df_combustibles, aes(x,y), colour="purple", size=0.5) +
      geom_point(data=df_hors_zone, aes(x,y), colour="lightblue", alpha=0.3, size=0.35) +
      coord_fixed(xlim=c(0,1), ylim=c(0,1), expand=FALSE) +
      theme_minimal() +
      labs(
         title=paste("Jour", jour),
         subtitle=paste0("Aire : ", round(fraction_aire_cumulee * 100, 1), 
                          "% | Front Int : ", round(frontiere_interne_totale, 2),
                          " | Front Ext : ", round(frontiere_externe_totale, 2))
      )
   
   print(p)
   Sys.sleep(0.4)
   
   # Arrêt si le carré est couvert
   if(fraction_aire_cumulee >= 0.999){
      cat("Le carré est entièrement brûlé.\n")
      break
   }
   
   # Passage au lendemain
   if(pts_comb_du_jour$n == 0){
      cat("Plus aucun nouveau combustible atteint.\n")
      break
   } else {
      pts_actuels <- pts_comb_du_jour
      jour <- jour + 1
   }
}

4.2 Probabilité que le feu s’éteigne après la première étape

Notons par \(h(n\vert x_0)\) la probabilité que le feu s’éteigne au plus à la n-ième étape, où \(x_0\) est l’ensemble des points qui ont été générés à l’intérieur de la réunion des feux du jour précédent. Alors \(h(1\vert x_0)\) représente la probabilité qu’aucun combustible ne soit généré au jour 2 dans \(\delta(x_0)\). On a \(h(1\vert x_0)=exp(-\theta(\delta(x_0)))\)(Serra et al. 2010).

   #------------------------------------------------------

```{r}
   zone_combustible_restant <- setminus.owin(w, union_totale)
   
   # Processus de Poisson généré UNIQUEMENT dans la zone non brûlée
   pts_comb_tous <- rpoispp(theta, win = zone_combustible_restant)
```

## Probabilité que le feu s'éteigne après la première étape

Notons par $h(n\vert x_0)$ la probabilité que le feu s'éteigne au plus à la n-ième étape, où $x_0$ est l'ensemble des points qui ont été générés à l'intérieur de la réunion des feux du jour précédent. Alors $h(1\vert x_0)$ représente la probabilité qu'aucun combustible ne soit généré au jour 2 dans $\delta(x_0)$. On a $h(1\vert x_0)=exp(-\theta(\delta(x_0)))$[@serra2010].

On obtient $\hat{\theta}(x,y)\approx 34,84x+31,79y +69,30$. Pour $\hat\delta$, on estime de la même manière qu'au premier exemple, on obtient $\hat\delta=0,04$. Il ne reste plus qu'à faire les calculs; $\mathbb{E}(N(\hat\delta(x_0)))= \int_{\hat\delta(x_0)}\hat\theta(x,y)dA=\pi r^2(\hat\theta (0,1;0,4)+ \hat\theta(0,4;0,3) +\hat\theta(0,7;0,2)+ \hat\theta(0,8;0,8))= \pi(0,04)^2(88,68+92,77+100,05+122,6)\approx2,03$, de telle sorte que $N(\hat\delta(x_0))\sim Poisson(2,03)$ et donc, $\mathbb{P}(N(\hat\delta(x_0))\geq 1)= 1-\mathbb{P} (N(\hat\delta(x_0))=0) =1=e^{-2,03}\approx0,87$. Faisons la simulation:

```{r}

# --- FONCTION THETA LINÉAIRE ax + by + c ---

  50 * x + 50 * y + 50

for(sim in 1:n_simulations) {
  
  # Génération d'une nouvelle grille de combustibles (Poisson non homogène via theta)
  pts_comb_tous <- rpoispp(theta, win = w)
  
  # Détection des captures
  est_touche <- inside.owin(pts_comb_tous$x, pts_comb_tous$y, union_jour1)
  nb_touches <- sum(est_touche)
  
  historique_comptes[sim] <- nb_touches

  }
}
```
